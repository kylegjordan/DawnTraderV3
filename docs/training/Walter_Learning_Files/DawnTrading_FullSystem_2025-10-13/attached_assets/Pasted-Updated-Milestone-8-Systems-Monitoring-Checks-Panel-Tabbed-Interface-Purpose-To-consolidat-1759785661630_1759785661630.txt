Updated Milestone #8: Systems Monitoring & Checks Panel (Tabbed Interface)
🧭 Purpose

To consolidate all system-level tools, logs, and reports into a single panel with tabs — separating this functionality from AI Analysis and Settings.

This panel becomes the command center for monitoring, validating, and auditing the AI and trading subsystems.

🗂️ Panel Overview
Panel Name:

Systems Monitoring & Checks

Tabs Included:
Tab	Source	Purpose
1️⃣ System Health	/api/system/health	Shows status of trading engine, AI modules, data feeds, uptime, and cron jobs.
2️⃣ Audit Viewer	strategy_settings_audit table	Lists all strategy parameter changes (user or AI).
3️⃣ System Logs	globalThis.systemLogs or server/services/system-logger.ts	Displays engine and app logs.
4️⃣ Validation Reports	(moved from AI Analysis panel)	Shows AI model and data validation results.
5️⃣ Audit Log	(moved from AI Analysis panel)	Detailed AI operation audit (e.g. decisions made, rejected trades).
6️⃣ Error Logs	(moved from AI Analysis panel)	Centralized view of backend or AI-related runtime errors.

2️⃣ Frontend Component

File: client/src/components/system/system-monitoring-panel.tsx

Use a tabbed layout with shadcn/ui Tabs.

import React, { useEffect, useState } from "react";
import { Tabs, TabsList, TabsTrigger, TabsContent } from "@/components/ui/tabs";
import { Card, CardContent, CardHeader } from "@/components/ui/card";
import { apiRequest } from "@/lib/api";

export default function SystemMonitoringPanel() {
  const [health, setHealth] = useState(null);
  const [audits, setAudits] = useState([]);
  const [logs, setLogs] = useState([]);
  const [validationReports, setValidationReports] = useState([]);
  const [aiAuditLog, setAiAuditLog] = useState([]);
  const [errorLogs, setErrorLogs] = useState([]);

  useEffect(() => {
    apiRequest("GET", "/api/system/health").then(r => setHealth(r.status));
    apiRequest("GET", "/api/strategies/audit").then(r => setAudits(r.audits));
    apiRequest("GET", "/api/system/logs").then(r => setLogs(r.logs));
    apiRequest("GET", "/api/system/validation-reports").then(r => setValidationReports(r.reports));
    apiRequest("GET", "/api/system/ai-audit-log").then(r => setAiAuditLog(r.logs));
    apiRequest("GET", "/api/system/error-logs").then(r => setErrorLogs(r.errors));
  }, []);

  return (
    <Card className="mt-8">
      <CardHeader>Systems Monitoring & Checks</CardHeader>
      <CardContent>
        <Tabs defaultValue="health" className="w-full">
          <TabsList className="flex flex-wrap">
            <TabsTrigger value="health">System Health</TabsTrigger>
            <TabsTrigger value="audit">Audit Viewer</TabsTrigger>
            <TabsTrigger value="logs">System Logs</TabsTrigger>
            <TabsTrigger value="validation">Validation Reports</TabsTrigger>
            <TabsTrigger value="aiAudit">Audit Log</TabsTrigger>
            <TabsTrigger value="errors">Error Logs</TabsTrigger>
          </TabsList>

          {/* System Health */}
          <TabsContent value="health">
            {health ? (
              <ul className="mt-2">
                {Object.entries(health).map(([k, v]) => (
                  <li key={k}><strong>{k}</strong>: {String(v)}</li>
                ))}
              </ul>
            ) : <p>Loading...</p>}
          </TabsContent>

          {/* Audit Viewer */}
          <TabsContent value="audit">
            {audits.length ? audits.map(a => (
              <div key={a.id} className="border-b py-1 text-sm">
                <strong>{a.strategy}</strong> ({a.mode}) — {a.actorType} at {new Date(a.createdAt).toLocaleString()}
                <br />
                <small>{Object.keys(a.changes || {}).map(k => `${k}: ${a.changes[k].old} → ${a.changes[k].new}`).join(", ")}</small>
              </div>
            )) : <p>No audit entries found.</p>}
          </TabsContent>

          {/* System Logs */}
          <TabsContent value="logs">
            <pre className="text-xs bg-gray-900 text-white p-2 rounded overflow-auto h-64">
              {logs.map(l => `[${l.timestamp}] ${l.message}`).join("\n")}
            </pre>
          </TabsContent>

          {/* Validation Reports */}
          <TabsContent value="validation">
            {validationReports.length ? validationReports.map(r => (
              <div key={r.id} className="border-b py-1 text-sm">
                {r.timestamp} — {r.summary}
              </div>
            )) : <p>No validation reports available.</p>}
          </TabsContent>

          {/* AI Audit Log */}
          <TabsContent value="aiAudit">
            {aiAuditLog.length ? aiAuditLog.map(l => (
              <div key={l.id} className="border-b py-1 text-sm">
                {l.timestamp} — {l.action}: {l.details}
              </div>
            )) : <p>No AI audit log entries available.</p>}
          </TabsContent>

          {/* Error Logs */}
          <TabsContent value="errors">
            <pre className="text-xs bg-red-900 text-white p-2 rounded overflow-auto h-64">
              {errorLogs.map(e => `[${e.timestamp}] ${e.message}`).join("\n")}
            </pre>
          </TabsContent>
        </Tabs>
      </CardContent>
    </Card>
  );
}
