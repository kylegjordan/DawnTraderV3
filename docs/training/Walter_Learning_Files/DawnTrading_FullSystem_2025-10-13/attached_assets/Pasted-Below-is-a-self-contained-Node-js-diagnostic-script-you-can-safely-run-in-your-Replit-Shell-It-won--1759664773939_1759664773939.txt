Below is a self-contained Node.js diagnostic script you can safely run in your Replit Shell. It won’t perform any trades — it just builds and logs the exact signature and response for the /0/private/Balance endpoint.

🧩 File: testKrakenSignature.js
/**
 * Kraken API Signature Debugging Script
 * -------------------------------------
 * This script checks whether your Kraken API key and secret
 * generate a valid authenticated signature for the Balance endpoint.
 *
 * Usage:
 *   node testKrakenSignature.js
 */

import crypto from "crypto";
import fetch from "node-fetch";

// --- Load API credentials from environment ---
const API_KEY = process.env.KRAKEN_API_KEY;
const API_SECRET = process.env.KRAKEN_API_SECRET;

if (!API_KEY || !API_SECRET) {
  console.error("❌ Missing environment variables KRAKEN_API_KEY or KRAKEN_API_SECRET");
  process.exit(1);
}

// --- Kraken REST API endpoint ---
const KRAKEN_URL = "https://api.kraken.com";
const PATH = "/0/private/Balance";

// --- Generate nonce ---
const nonce = Date.now() * 1000; // microsecond-style nonce
const postData = new URLSearchParams({ nonce: nonce.toString() }).toString();

// --- Step 1: SHA256(nonce + postData) ---
const message = nonce + postData;
const hash = crypto.createHash("sha256").update(message).digest();

// --- Step 2: Concatenate path + hash ---
const hmacMessage = Buffer.concat([Buffer.from(PATH), hash]);

// --- Step 3: HMAC-SHA512 over base64-decoded secret ---
const hmac = crypto
  .createHmac("sha512", Buffer.from(API_SECRET, "base64"))
  .update(hmacMessage)
  .digest("base64");

// --- Step 4: Show debugging info ---
console.log("🧠 Debugging Signature Generation:");
console.log("---------------------------------");
console.log("Nonce:", nonce);
console.log("Post Data:", postData);
console.log("SHA256(nonce+postData):", hash.toString("hex"));
console.log("HMAC Message (path+hash):", hmacMessage.toString("hex"));
console.log("HMAC Signature (Base64):", hmac);
console.log("\n🔑 API Key (first 8 chars):", API_KEY.slice(0, 8) + "...");
console.log("---------------------------------\n");

// --- Step 5: Send the request ---
(async () => {
  try {
    const res = await fetch(KRAKEN_URL + PATH, {
      method: "POST",
      headers: {
        "API-Key": API_KEY,
        "API-Sign": hmac,
        "Content-Type": "application/x-www-form-urlencoded",
      },
      body: postData,
    });

    const data = await res.json();
    console.log("📡 Kraken Response:\n", JSON.stringify(data, null, 2));

    if (data.error && data.error.length) {
      console.log("\n⚠️ Kraken returned error:", data.error.join(", "));
      if (data.error.includes("EAPI:Invalid key")) {
        console.log("\n💡 This means Kraken rejected your credentials or signature.");
        console.log("   Next steps:");
        console.log("   - Confirm your key pair is from the same API key in Kraken.");
        console.log("   - Check for extra spaces/newlines in the secret.");
        console.log("   - Ensure 'Query Funds' permission is enabled.");
      }
    } else {
      console.log("\n✅ Success! Your keys and signature are valid.");
    }
  } catch (err) {
    console.error("❌ Network or fetch error:", err);
  }
})();

🧠 How to Run It

In your Replit project, create a new file named testKrakenSignature.js.

Paste in the code above.

Ensure your Replit Secrets contain:

KRAKEN_API_KEY
KRAKEN_API_SECRET


(Use your newest Kraken key pair.)

Open the Shell tab in Replit and run:

node testKrakenSignature.js


Watch the output — it will show:

Nonce

SHA-256 hash of payload

Full HMAC signature

Kraken’s API response

🧾 What to Expect
Result	Meaning
{ "result": { ... } }	✅ Keys and signature are valid.
"EGeneral:Temporary lockout"	✅ Keys are valid, but rate-limited (wait a few minutes).
"EAPI:Invalid key"	❌ Keys or signature invalid (we’ll compare the computed HMAC to Kraken’s expected pattern).
"EGeneral:Permission denied"	The key lacks “Query Funds” permission.
