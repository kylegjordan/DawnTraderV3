2.2 Strengthen Finnhub / Fallback Logic

Intent: Reliable, layered data retrieval (Finnhub → CoinGecko → fallback cache).

Replit: Update server/services/finnhub-service.ts

// server/services/finnhub-service.ts
import axios from "axios";
import NodeCache from "node-cache";

const cache = new NodeCache({ stdTTL: 300 }); // 5-min cache

const FINNHUB_KEY = process.env.FINNHUB_API_KEY!;
const BASE = "https://finnhub.io/api/v1";

export async function searchSymbol(query: string) {
  if (!query) return [];
  const cacheKey = `symbol:${query.toUpperCase()}`;
  const cached = cache.get(cacheKey);
  if (cached) return cached;

  try {
    const url = `${BASE}/search?q=${encodeURIComponent(query)}&token=${FINNHUB_KEY}`;
    const { data } = await axios.get(url);
    cache.set(cacheKey, data.result);
    return data.result || [];
  } catch (err) {
    console.error("Finnhub search error:", err.message);
    return [];
  }
}

export async function getSymbolQuote(symbol: string) {
  const cacheKey = `quote:${symbol}`;
  const cached = cache.get(cacheKey);
  if (cached) return cached;

  try {
    const url = `${BASE}/quote?symbol=${encodeURIComponent(symbol)}&token=${FINNHUB_KEY}`;
    const { data } = await axios.get(url);
    if (!data || !data.c) throw new Error("Invalid data");
    cache.set(cacheKey, data);
    return data;
  } catch (err) {
    console.warn(`Finnhub quote fallback for ${symbol}:`, err.message);
    // fallback to CoinGecko (crypto only)
    try {
      const { data } = await axios.get(`https://api.coingecko.com/api/v3/simple/price?ids=${symbol.toLowerCase()}&vs_currencies=usd`);
      return { c: data?.[symbol.toLowerCase()]?.usd ?? null };
    } catch {
      return null;
    }
  }
}

2.3 Improve API Endpoints

Replit: Adjust routes in server/routes.ts

// server/routes.ts
import { searchSymbol, getSymbolQuote } from "./services/finnhub-service";

// Search endpoint
app.get("/api/symbol/search", async (req, res) => {
  const q = (req.query.q || "").toString().trim();
  if (!q) return res.status(400).json({ error: "Query required" });

  const result = await searchSymbol(q);
  if (!result || result.length === 0) return res.status(404).json({ message: "No results found" });
  res.json(result);
});

// Quote endpoint
app.get("/api/symbol/:symbol/quote", async (req, res) => {
  const { symbol } = req.params;
  const quote = await getSymbolQuote(symbol);
  if (!quote) return res.status(404).json({ message: "No quote data available" });
  res.json(quote);
});

2.4 Frontend Component Fix

Replit: Update the component that renders the search results.

// client/src/components/symbol-search.tsx
import { useState } from "react";
import { apiRequest } from "@/lib/query";

export default function SymbolSearch() {
  const [query, setQuery] = useState("");
  const [results, setResults] = useState<any[]>([]);
  const [loading, setLoading] = useState(false);

  async function handleSearch() {
    setLoading(true);
    try {
      const res = await apiRequest(`/api/symbol/search?q=${query}`);
      setResults(res);
    } catch (err) {
      setResults([]);
      console.error(err);
    } finally {
      setLoading(false);
    }
  }

  return (
    <div className="space-y-2">
      <input
        type="text"
        placeholder="Search by symbol or company name…"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        className="border px-3 py-2 rounded w-full"
      />
      <button onClick={handleSearch} className="bg-blue-600 text-white px-4 py-2 rounded">
        {loading ? "Searching..." : "Search"}
      </button>
      {results.length > 0 && (
        <ul className="mt-3 divide-y">
          {results.map((r) => (
            <li key={r.symbol} className="py-2">
              <div className="font-semibold">{r.symbol}</div>
              <div className="text-sm text-gray-500">{r.description}</div>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}
