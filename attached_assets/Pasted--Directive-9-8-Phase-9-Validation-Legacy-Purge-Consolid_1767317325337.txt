🧪 Validation Script (verify-phase-9.ts)
import { storage } from './server/storage';
import { SYSTEM_GUARDS } from './server/config/system-guards';

async function verify() {
  console.log("1. Checking Guardrails V2 Access...");
  const v2 = await storage.getGuardrailsV2({ mode: 'paper' });
  if (v2.symbolCooldownMinutes !== 5) throw new Error("Guardrails V2 mismatch!");

  console.log("2. Checking Legacy Block...");
  try {
    // @ts-ignore
    await storage.getGuardrails({ mode: 'paper' });
    throw new Error("Legacy getGuardrails() did not throw!");
  } catch {
    console.log("   ✅ Legacy access correctly blocked.");
  }

  console.log("3. Checking SYSTEM_GUARDS Consistency...");
  if (SYSTEM_GUARDS.MIN_LIQUIDITY_SCORE !== 40 || SYSTEM_GUARDS.MAX_VOL_NOISE !== 0.6)
    throw new Error("SYSTEM_GUARDS mismatch!");

  console.log("4. Checking ActiveFilterPool Consistency...");
  const active = await storage.getActivePool('paper');
  if (!Array.isArray(active)) throw new Error("Active Filter Pool missing or corrupt.");

  console.log("✅ SYSTEM INTEGRITY CONFIRMED");
}
verify().catch(console.error);
