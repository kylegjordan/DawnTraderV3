1️⃣ Create Adaptive Profiler Core

File: server/services/gemini-adaptive-profiler.ts

import { cacheStats, setCache } from "./cache";
import { recordTradeMetric } from "./telemetry-service";

let statsHistory: any[] = [];

export async function geminiAdaptiveTick() {
  const snapshot = {
    ts: Date.now(),
    cacheHit: cacheStats().ratio,
    latency: global.avgLatency || 0,
    cpu: process.cpuUsage().user / 1e6,
    mem: process.memoryUsage().rss / 1024 / 1024,
  };
  statsHistory.push(snapshot);
  if (statsHistory.length > 30) statsHistory.shift();

  // === Adaptive Logic ===
  if (snapshot.cacheHit < 0.8) process.env.DEFAULT_CACHE_TTL = "120000";
  else if (snapshot.cacheHit > 0.9) process.env.DEFAULT_CACHE_TTL = "60000";

  const avgLatency = statsHistory.map(s => s.latency).reduce((a,b)=>a+b,0)/statsHistory.length || 0;
  if (avgLatency > 140) process.env.TELEMETRY_BATCH_MS = "3000";
  else process.env.TELEMETRY_BATCH_MS = "2000";

  console.log(`[Gemini-Adaptive] cache=${snapshot.cacheHit.toFixed(2)} ttl=${process.env.DEFAULT_CACHE_TTL} batch=${process.env.TELEMETRY_BATCH_MS}`);
  recordTradeMetric({ metric:"adaptive_tick", ...snapshot });
}
setInterval(geminiAdaptiveTick, 60000);


✅ Expected: A new [Gemini-Adaptive] log line every 60 s with cache ratio and adaptive TTL adjustments.
