2. Backend Implementation

Create: server/services/config-service.ts

import db from "../db";
import { logger } from "../utils/logger";

export class ConfigService {
  static async getAll() {
    const { rows } = await db.query("SELECT key, value, type, updated_at, updated_by FROM config_registry ORDER BY key");
    return rows;
  }

  static async get(key: string) {
    const { rows } = await db.query("SELECT * FROM config_registry WHERE key=$1", [key]);
    return rows[0];
  }

  static async update(key: string, value: any, type: string, user: string) {
    await db.query(
      `INSERT INTO config_registry (key, value, type, updated_by)
       VALUES ($1,$2,$3,$4)
       ON CONFLICT (key) DO UPDATE
       SET value=EXCLUDED.value, type=EXCLUDED.type, updated_at=NOW(), updated_by=EXCLUDED.updated_by`,
      [key, value, type, user]
    );
    logger.info("Config updated", { key, user });
  }
}

4. API Endpoints

Edit server/routes.ts:

import { ConfigService } from "../services/config-service";
import { requireAdmin } from "../middleware/auth";

router.get("/api/config", async (req, res) => {
  res.json(await ConfigService.getAll());
});

router.put("/api/config", requireAdmin, async (req, res) => {
  const { key, value, type } = req.body;
  const user = req.user?.username ?? "system";
  await ConfigService.update(key, value, type, user);
  res.json({ ok: true });
});

5. Frontend UI Panel

Create client/pages/SystemConfig.tsx

import React, { useEffect, useState } from "react";
import { Card, CardContent } from "@/components/ui/card";
import { Button } from "@/components/ui/button";

export default function SystemConfig() {
  const [configs, setConfigs] = useState([]);

  async function loadConfigs() {
    const res = await fetch("/api/config");
    setConfigs(await res.json());
  }

  async function updateConfig(key, value, type) {
    await fetch("/api/config", {
      method: "PUT",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ key, value, type })
    });
    await loadConfigs();
  }

  useEffect(() => { loadConfigs(); }, []);

  return (
    <div className="p-6 space-y-4">
      {configs.map(cfg => (
        <Card key={cfg.key} className="shadow-md">
          <CardContent className="flex justify-between items-center">
            <div>
              <p className="font-semibold">{cfg.key}</p>
              <p className="text-xs text-gray-500">{cfg.type}</p>
            </div>
            <Button onClick={() => updateConfig(cfg.key, cfg.value, cfg.type)}>Update</Button>
          </CardContent>
        </Card>
      ))}
    </div>
  );
}

Add route in UI router: /system/config (admin-only).
