2️⃣ Twin Test — server/tests/integration/parity.test.ts
import { SYSTEM_GUARDS } from "@/config/system-guards";
import { TradeExecutor } from "@/services/live-trade-executor";
import { PaperExecutionEngine } from "@/services/paper-execution-engine";
import { SymbolMappingService } from "@/utils/symbol-mapping";

describe("Directive 9.6 — Sim-to-Live Parity", () => {
  test("Live vs VTS identical under identical tick", async () => {
    const symbol = SymbolMappingService.resolve("BTC/USD");
    const tick = { price: 50000, DI: 80, Noise: 0.2, correlation: 0.1 };

    const liveResult = await TradeExecutor.evaluate(symbol, tick);
    const simResult  = await PaperExecutionEngine.evaluate(symbol, tick);

    expect(Math.abs(liveResult.entry - simResult.entry))
      .toBeLessThan(SYSTEM_GUARDS.PARITY_TOLERANCE);
    expect(Math.abs(liveResult.stop - simResult.stop))
      .toBeLessThan(SYSTEM_GUARDS.PARITY_TOLERANCE);
    expect(Math.abs(liveResult.qty - simResult.qty))
      .toBeLessThan(SYSTEM_GUARDS.PARITY_TOLERANCE);
    expect(Math.abs(liveResult.ev - simResult.ev))
      .toBeLessThan(SYSTEM_GUARDS.PARITY_TOLERANCE);
    expect(liveResult.cwqiScore)
      .toBeCloseTo(simResult.cwqiScore, 5);
  });
});
