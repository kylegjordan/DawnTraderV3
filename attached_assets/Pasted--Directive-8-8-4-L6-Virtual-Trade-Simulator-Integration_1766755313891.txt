Simulator Logic
const feeRate = 0.0026;  // Kraken per side
const avgSlippage = 0.0015;
const tradeDuration = 3 * 60 * 60 * 1000; // 3 hours window

simulateTrade(signal) {
  const entry = signal.entryPrice;
  const tp = signal.takeProfit;
  const sl = signal.stopLoss;
  const spread = signal.spread;
  const outcome = getMarketOutcome(signal.symbol, tradeDuration);

  let exitPrice;
  let resultType;

  if (outcome.high >= tp) {
    exitPrice = tp;
    resultType = "take_profit";
  } else if (outcome.low <= sl) {
    exitPrice = sl;
    resultType = "stop_loss";
  } else {
    exitPrice = outcome.close;
    resultType = "timeout";
  }

  const gross = (exitPrice - entry) / entry;
  const fees = entry * (feeRate * 2) + entry * avgSlippage;
  const net = gross - (fees / entry);

  return { ...signal, resultType, gross, net };
}
