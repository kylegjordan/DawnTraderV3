Task 2 — In-Memory Cost Cache

File: /server/core/cache/cost-cache.ts





const cache = new Map();export function getCostMetrics(symbol: string) {

const item = cache.get(symbol);

if (item && Date.now() - item.t < 60_000) return item.v;

return null;

}export function setCostMetrics(symbol: string, data) {

const clamped = {

fee: Math.min(data.fee, MAX_COST_BOUND),

slippage: Math.min(data.slippage, MAX_COST_BOUND),

spread: Math.min(data.spread, MAX_COST_BOUND)

};

cache.set(symbol, { v: clamped, t: Date.now() });

return clamped;

}

✅ TTL = 60 s.



✅ Pure in-memory (nanosecond lookups).



✅ Clamps values to safe ranges.
