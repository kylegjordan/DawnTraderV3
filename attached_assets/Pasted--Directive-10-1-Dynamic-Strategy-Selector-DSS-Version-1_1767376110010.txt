The DSS introduces intelligence and restraint to the system:
it enforces physics-first logic, blocks trades in Extreme Noise markets, and routes execution dynamically to the best-fitting strategy for current conditions.

2️⃣ Implement DSS Service (server/services/dynamic-strategy-selector.ts)
import { SYSTEM_GUARDS } from '../config/system-guards';
import { calculateVolNoise, calculateTrendSlope } from '../utils/analysis-utils';

export class DynamicStrategySelector {
  determineRegime(metrics) {
    const vol = metrics.volNoise;
    const trend = metrics.trendSlope;

    // 1️⃣ Extreme Noise: Auto-Veto
    if (vol > SYSTEM_GUARDS.MAX_VOL_NOISE) return 'EXTREME_NOISE';

    // 2️⃣ Bull Regimes
    if (trend > SYSTEM_GUARDS.REGIME_THRESHOLDS.TREND_POSITIVE) {
      return vol < SYSTEM_GUARDS.REGIME_THRESHOLDS.VOL_LOW
        ? 'BULL_STABLE'
        : 'BULL_VOLATILE';
    }

    // 3️⃣ Bear Regimes
    if (trend < -SYSTEM_GUARDS.REGIME_THRESHOLDS.TREND_POSITIVE) {
      return vol < SYSTEM_GUARDS.REGIME_THRESHOLDS.VOL_LOW
        ? 'BEAR_STABLE'
        : 'BEAR_VOLATILE';
    }

    // 4️⃣ Flat / Sideways Regime
    return 'LOW_VOL_CHOP';
  }

  evaluate(snapshot, strategyMetrics) {
    const regime = this.determineRegime(snapshot);

    if (regime === 'EXTREME_NOISE')
      return { veto: true, reason: 'Extreme Volatility (>0.6)', regime };

    const candidates = SYSTEM_GUARDS.STRATEGY_MAP[regime] || [];
    const viable = candidates.filter(s => strategyMetrics[s]?.netEV > 0);

    if (!viable.length)
      return { veto: true, reason: 'No positive NetEV strategy', regime };

    const best = viable.sort(
      (a, b) =>
        (strategyMetrics[b]?.confidence || 0) -
        (strategyMetrics[a]?.confidence || 0)
    )[0];

    return { regime, strategy: best, confidence: strategyMetrics[best]?.confidence || 0 };
  }
}


✅ Handles all five regimes including LOW_VOL_CHOP.
✅ Returns explicit veto under EXTREME_NOISE and NetEV ≤ 0.
