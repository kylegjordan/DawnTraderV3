1️⃣ Purpose & Overview

The goal of this feature is to secure the trading application by requiring authenticated logins for all users accessing the dashboard, whether through desktop or mobile (PWA).

Users must log in using a strong password and may optionally authenticate with biometric verification (fingerprint or Face ID) if supported by their device.

This authentication will:

Protect access to trading data and API keys

Maintain persistent sessions using JWT tokens

Require reauthentication on session expiry

Work seamlessly in both live and paper trading modes

Support biometric unlock for installed mobile PWAs

2.2 Password Policy Enforcement

When users register or update passwords, enforce the following:

Minimum 8 characters

At least 1 uppercase, 1 number, and 1 special character

Reject common or weak passwords


File: server/services/auth-service.ts

import bcrypt from "bcryptjs";

export function validatePasswordStrength(password: string): boolean {
  return /^(?=.*[A-Z])(?=.*\d)(?=.*[!@#$%^&*()_+=\-{};:'",.<>?])(?=.{8,})/.test(password);
}

export async function hashPassword(password: string): Promise<string> {
  return bcrypt.hash(password, 10);
}

export async function verifyPassword(password: string, hash: string): Promise<boolean> {
  return bcrypt.compare(password, hash);
}

2.3 Auth API Routes

File: server/routes.ts

import jwt from "jsonwebtoken";
import { validatePasswordStrength, hashPassword, verifyPassword } from "./services/auth-service";
import storage from "./storage";

const JWT_SECRET = process.env.JWT_SECRET || "change_this_secret";

// REGISTER
app.post("/api/auth/register", async (req, res) => {
  const { username, password } = req.body;
  if (!validatePasswordStrength(password))
    return res.status(400).json({ error: "Weak password" });
  const hash = await hashPassword(password);
  await storage.createUser({ username, passwordHash: hash });
  res.json({ success: true });
});

// LOGIN
app.post("/api/auth/login", async (req, res) => {
  const { username, password } = req.body;
  const user = await storage.getUserByUsername(username);
  if (!user) return res.status(404).json({ error: "User not found" });
  const valid = await verifyPassword(password, user.passwordHash);
  if (!valid) return res.status(401).json({ error: "Invalid credentials" });
  const token = jwt.sign({ id: user.id, username }, JWT_SECRET, { expiresIn: "12h" });
  res.json({ token });
});

// VERIFY TOKEN
app.get("/api/auth/verify", async (req, res) => {
  try {
    const authHeader = req.headers.authorization;
    if (!authHeader) throw new Error("Missing header");
    const token = authHeader.split(" ")[1];
    const decoded = jwt.verify(token, JWT_SECRET);
    res.json({ valid: true, user: decoded });
  } catch {
    res.status(401).json({ valid: false });
  }
});

2.5 Frontend Login Page

File: client/src/pages/login.tsx

import { useState } from "react";
import { apiRequest } from "@/lib/query";
import { Button } from "@/components/ui/button";

export default function LoginPage() {
  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");

  async function handleLogin() {
    try {
      const res = await apiRequest("/api/auth/login", "POST", { username, password });
      localStorage.setItem("token", res.token);
      window.location.href = "/dashboard";
    } catch (err: any) {
      setError(err?.message || "Login failed");
    }
  }

  return (
    <div className="flex h-screen items-center justify-center bg-gray-50">
      <div className="p-6 bg-white rounded-xl shadow-lg w-80">
        <h2 className="text-xl font-semibold text-center mb-4">Login</h2>
        <input className="w-full mb-3 p-2 border rounded" placeholder="Username" value={username} onChange={e => setUsername(e.target.value)} />
        <input className="w-full mb-3 p-2 border rounded" placeholder="Password" type="password" value={password} onChange={e => setPassword(e.target.value)} />
        {error && <p className="text-red-500 text-sm mb-3">{error}</p>}
        <Button className="w-full" onClick={handleLogin}>Sign In</Button>
      </div>
    </div>
  );
}

export async function attemptBiometricLogin() {
  if (!("credentials" in navigator)) return null;
  const cred = await navigator.credentials.get({ publicKey: { userVerification: "required" } });
  if (cred) {
    // Send to server for validation (future extension)
    return true;
  }
  return false;
}

Then call attemptBiometricLogin() on mobile startup to allow auto-login via biometrics if the session is expired.
