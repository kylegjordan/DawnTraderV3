3. Implement Decay Logic

File: server/services/hybrid-integration.ts

Add the decay method inside HybridIntegrationService:

import { HYBRID_PARAMS } from '@/server/config/system-guards';

private applyPatternDecay(originalStrength: number, deltaCandles: number): number {
  const { ENABLED, LAMBDA, FLOOR } = HYBRID_PARAMS.DECAY;
  if (!ENABLED) return originalStrength;
  const decayed = originalStrength * Math.exp(-LAMBDA * deltaCandles);
  return Math.max(decayed, originalStrength * FLOOR);
}


Then, in detectConfluence():

Critical fix — normalize timestamp difference:

// Convert time delta from ms → candle units (assuming 1-hour candles)
const deltaCandles = Math.abs(q.timestamp - p.timestamp) / 3600000;
const effectiveStrength = this.applyPatternDecay(p.strength, deltaCandles);


Use effectiveStrength in ensemble computation:

const hybridScore = this.computeEnsembleScore(
  q.expectancy ?? 0.5,
  effectiveStrength,
  p.predictiveConfidence

5. Verification – Unit Tests

File: server/tests/unit/hybrid-integration.test.ts

Add suite: Pattern Decay Logic

describe('Pattern Decay Logic', () => {
  test('Fresh Pattern (Age 0)', () => {
    expect(applyPatternDecay(1.0, 0)).toBeCloseTo(1.0);
  });

  test('Aged Pattern (3 candles)', () => {
    const result = applyPatternDecay(1.0, 3);
    expect(result).toBeCloseTo(0.65, 1); // ~65% of original
  });

  test('Floor Enforcement (Age 50)', () => {
    const result = applyPatternDecay(1.0, 50);
    expect(result).toBeGreaterThanOrEqual(0.3);
  });

  test('Disabled Decay', () => {
    HYBRID_PARAMS.DECAY.ENABLED = false;
    expect(applyPatternDecay(1.0, 5)).toBe(1.0);
  });
});
