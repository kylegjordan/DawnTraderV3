C) Coherency Analytics View → v_guardrails_compliance.sql

Create SQL view /server/db/views/v_guardrails_compliance.sql:

CREATE OR REPLACE VIEW v_guardrails_compliance AS
SELECT
  mode,
  portfolio_risk_per_trade_pct,
  daily_loss_kill_switch_pct,
  max_open_positions,
  symbol_cooldown_minutes,
  CASE
    WHEN portfolio_risk_per_trade_pct <= daily_loss_kill_switch_pct / 10 THEN 'PASS'
    WHEN portfolio_risk_per_trade_pct <= daily_loss_kill_switch_pct / 5 THEN 'WARN'
    ELSE 'FAIL'
  END AS coherency_status,
  is_manual_override,
  tuned_by_latti,
  locked_by_user,
  last_updated
FROM guardrails_v2;


Use this view to feed coherency analytics in the dashboard widget.

Add corresponding API endpoint /api/analytics/guardrails-compliance returning the aggregated status for display.
