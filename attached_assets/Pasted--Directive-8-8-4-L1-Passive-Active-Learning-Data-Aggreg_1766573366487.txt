1️⃣ Create Data Aggregator Service

File: server/services/data-aggregator.ts

import { EventEmitter } from 'events';
import fs from 'fs/promises';
import path from 'path';

export class DataAggregator extends EventEmitter {
  private buffer: any[] = [];
  private mode: 'passive' | 'active' = 'passive';
  private lastFlush = 0;
  private flushInterval = 30000; // 30s
  private aggregateInterval = 900000; // 15m
  private aggregateTimer: NodeJS.Timeout | null = null;

  constructor() {
    super();
    this.aggregateTimer = setInterval(() => this.aggregate(), this.aggregateInterval);
    setInterval(() => this.flush(), this.flushInterval);
  }

  setMode(mode: 'passive' | 'active') {
    this.mode = mode;
  }

  async capture(eventType: string, payload: Record<string, any>) {
    this.buffer.push({
      timestamp: Date.now(),
      mode: this.mode,
      eventType,
      ...payload
    });
  }

  async flush() {
    if (this.buffer.length === 0) return;
    const day = new Date().toISOString().slice(0,10);
    const dir = `./logs/data_aggregates`;
    await fs.mkdir(dir, { recursive: true });
    const filePath = path.join(dir, `${this.mode}_${day}.json`);
    const data = this.buffer.map(d => JSON.stringify(d)).join('\n') + '\n';
    await fs.appendFile(filePath, data);
    this.buffer = [];
    this.lastFlush = Date.now();
  }

  async aggregate() {
    const now = new Date();
    const periodKey = now.toISOString().slice(0,13);
    const srcDir = `./logs/data_aggregates`;
    const outDir = path.join(srcDir, 'hourly');
    await fs.mkdir(outDir, { recursive: true });
    const outFile = path.join(outDir, `${this.mode}_${periodKey}.json`);

    // Read today’s capture file
    const dayFile = path.join(srcDir, `${this.mode}_${now.toISOString().slice(0,10)}.json`);
    try {
      const content = await fs.readFile(dayFile, 'utf-8');
      const lines = content.trim().split('\n').map(l => JSON.parse(l));
      const byKey: Record<string, any[]> = {};
      for (const rec of lines) {
        const key = `${rec.symbol}_${rec.strategy}`;
        if (!byKey[key]) byKey[key] = [];
        byKey[key].push(rec);
      }

      const aggregates = Object.entries(byKey).map(([key, recs]) => {
        const [symbol, strategy] = key.split('_');
        const avg = (f: string) => recs.reduce((a, r) => a + (r[f] ?? 0), 0) / recs.length;
        return {
          timestamp: now.toISOString(),
          symbol,
          strategy,
          period: periodKey,
          avgNGC: avg('ngc'),
          avgCWQI: avg('cwqi'),
          avgRisk: avg('risk'),
          avgProfitRate: avg('profitRate'),
          reconfirmRate: avg('reconfirmed' as any),
          tradePromotionRate: avg('tradeExecuted' as any),
          sampleCount: recs.length
        };
      });

      // Global roll-up
      const global = {
        timestamp: now.toISOString(),
        avgNGC: this.avg(aggregates, 'avgNGC'),
        avgCWQI: this.avg(aggregates, 'avgCWQI'),
        avgRisk: this.avg(aggregates, 'avgRisk'),
        avgProfitRate: this.avg(aggregates, 'avgProfitRate'),
        totalSignals: aggregates.reduce((a, r) => a + r.sampleCount, 0)
      };

      await fs.writeFile(outFile, JSON.stringify({ aggregates, global }, null, 2));
    } catch (err) {
      console.error('[DataAggregator] aggregation failed', err);
    }
  }

  private avg(arr: any[], key: string) {
    if (arr.length === 0) return 0;
    return arr.reduce((a, r) => a + (r[key] ?? 0), 0) / arr.length;
  }
}

export const dataAggregator = new DataAggregator();
