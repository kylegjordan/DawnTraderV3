2️⃣ Relax & Log the Passive Debounce Guard

In server/services/trading-state-sync.ts:

let lastPassiveState = null;

function broadcastTradingState(payload) {
  if (payload.passiveLearning === lastPassiveState) {
    console.log('[Phase33B] Duplicate passiveLearning broadcast skipped');
    return;
  }
  lastPassiveState = payload.passiveLearning;
  setTimeout(() => { lastPassiveState = null; }, 2000); // ✅ 2-second reset window
  contextBridge.broadcast({ type: 'trading_state_changed', payload });
}
