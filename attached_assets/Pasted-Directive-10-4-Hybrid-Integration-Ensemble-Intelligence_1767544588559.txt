3. Create Hybrid Integration Service

File: server/services/hybrid-integration.ts

Create new class:

import { HYBRID_PARAMS } from '@/server/config/system-guards';
import { TradeSignal, PatternSignal, HybridStrategyType } from '@/server/types';

export class HybridIntegrationService {
  computeEnsembleScore(quantConf: number, patternStrength: number, mlConf?: number) {
    const w = HYBRID_PARAMS.WEIGHTS;
    return (
      quantConf * w.QUANT +
      patternStrength * w.PATTERN +
      (mlConf ?? 0.5) * w.PREDICTIVE
    );
  }

  detectConfluence(quantSignals: TradeSignal[], patternSignals: PatternSignal[]) {
    const hybrids: TradeSignal[] = [];

    for (const q of quantSignals) {
      const match = patternSignals.find(
        (p) =>
          p.symbol === q.symbol &&
          p.direction === q.direction && // Directional match required
          Math.abs(p.timestamp - q.timestamp) <= HYBRID_PARAMS.MAX_CONFLUENCE_WINDOW
      );

      if (!match) continue;

      const hybridScore = this.computeEnsembleScore(
        q.expectancy ?? 0.5,
        match.strength,
        match.predictiveConfidence
      );

      if (hybridScore < HYBRID_PARAMS.MIN_SCORE) continue;

      const hybridStrategy: HybridStrategyType =
        q.direction === 'BUY' ? 'H1_TREND_SNIPER' : 'H4_MOMENTUM_LINK';

      hybrids.push({
        ...q,
        signalType: 'HYBRID',
        hybridScore,
        hybridStrategy,
        patternType: match.pattern,
        patternStrength: match.strength,
        predictiveConfidence: match.predictiveConfidence,
        componentScores: {
          quant: q.expectancy ?? 0.5,
          pattern: match.strength,
          ml: match.predictiveConfidence ?? 0.5,
        },
      });

      console.log(
        `[10.4] HybridIntegration: HYBRID signal generated for ${q.symbol} | Score=${hybridScore.toFixed(
          3
        )} | Strategy=${hybridStrategy}`
      );
    }

    return hybrids;
  }
}
