cat > server/services/strategic-drive-service.ts <<'EOF'
import prisma from "../db";
import { getTelemetryForAllStrategies } from "./telemetry-service";

export class StrategicDriveService {
  async computeDriveMetrics(mode: "paper" | "live") {
    const telemetry = await getTelemetryForAllStrategies(mode);
    const results: any[] = [];
    for (const t of telemetry) {
      const driveScore =
        0.5 * t.hitRate +
        0.3 * t.avgRMultiple -
        0.2 * t.avgToxicity;
      results.push(
        await prisma.strategy_drive_metrics.create({
          data: {
            strategy: t.strategy,
            mode,
            totalProfitUSD: t.totalProfitUSD || 0,
            totalTrades: t.totalTrades || 0,
            winRate: t.hitRate || 0,
            avgRMultiple: t.avgRMultiple || 0,
            alphaStrength: t.alphaStrength || 0,
            riskExposure: t.avgToxicity || 0,
            driveScore,
          },
        })
      );
    }
    return results;
  }

  async summarizeDriveMetrics() {
    const metrics = await prisma.strategy_drive_metrics.findMany({
      orderBy: { timestamp: "desc" },
      take: 5,
    });
    if (metrics.length === 0) return;
    const grouped = metrics.reduce((acc: any, m) => {
      acc[m.strategy] = acc[m.strategy] || [];
      acc[m.strategy].push(m.driveScore);
      return acc;
    }, {});
    const avgScores: Record<string, number> = {};
    for (const k in grouped)
      avgScores[k] =
        grouped[k].reduce((a, b) => a + b, 0) / grouped[k].length;

    const entries = Object.entries(avgScores);
    const best = entries.sort((a, b) => b[1] - a[1])[0];
    const worst = entries.sort((a, b) => a[1] - b[1])[0];
    const globalSDI =
      entries.reduce((sum, [_, v]) => sum + v, 0) / entries.length;

    await prisma.strategy_drive_summary.create({
      data: {
        globalSDI,
        bestStrategy: best[0],
        weakestStrategy: worst[0],
        dhmaWeight: avgScores["dhma"] || 1,
        quantflowWeight: avgScores["quantflow"] || 1,
        trendpulseWeight: avgScores["trendpulse"] || 1,
        volsurfWeight: avgScores["volsurf"] || 1,
        momentumxWeight: avgScores["momentumx"] || 1,
      },
    });

    return { globalSDI, best, worst };
  }
}
EOF

// === Phase 31 – Strategic Drive Engine Startup ===
import { StrategicDriveService } from "./services/strategic-drive-service";
const strategicDrive = new StrategicDriveService();

// Run initial computation on startup and every 60 minutes
(async () => {
  console.log("[31.0][SDPOE] Strategic Drive Service initializing...");
  await strategicDrive.computeDriveMetrics("paper");
  await strategicDrive.summarizeDriveMetrics();
  setInterval(async () => {
    await strategicDrive.computeDriveMetrics("paper");
    await strategicDrive.summarizeDriveMetrics();
  }, 60 * 60 * 1000);
})();
