2. Introduce Central Price Cache Module
2.1 – New file: server/services/price-cache.ts

Create a small, in-memory singleton cache keyed by internal canonical symbol (e.g. "APT/EUR", "TRX/USD"):

// server/services/price-cache.ts

export type PriceSourceTag = 'kraken_ws' | 'kraken_rest';

export interface CachedPrice {
  symbol: string;                // internal / canonical symbol, e.g. "APT/EUR"
  price: number;
  lastSource: PriceSourceTag;
  lastUpdatedAt: number;         // Date.now() ms
}

class PriceCache {
  private prices = new Map<string, CachedPrice>();

  updateFromWebSocket(symbol: string, price: number) {
    const now = Date.now();
    this.prices.set(symbol, {
      symbol,
      price,
      lastSource: 'kraken_ws',
      lastUpdatedAt: now,
    });
  }

  updateFromRest(symbol: string, price: number) {
    const now = Date.now();
    this.prices.set(symbol, {
      symbol,
      price,
      lastSource: 'kraken_rest',
      lastUpdatedAt: now,
    });
  }

  get(symbol: string): CachedPrice | null {
    return this.prices.get(symbol) ?? null;
  }

  snapshot() {
    return Array.from(this.prices.values());
  }

  clear() {
    this.prices.clear();
  }
}

export const priceCache = new PriceCache();


Notes:

This cache is only for open positions in practice, because we will only call it from the WebSocket adapter and the price adapters that are already used by the paper engine.

No persistence. In-memory only.

7.2 – New diagnostic endpoint: /api/diagnostics/ia-price-cache/status

In server/routes.ts, add:

router.get('/api/diagnostics/ia-price-cache/status', async (req, res) => {
  try {
    const snapshot = priceCache.snapshot();
    const now = Date.now();

    const summary = {
      totalSymbols: snapshot.length,
      bySource: snapshot.reduce((acc, p) => {
        acc[p.lastSource] = (acc[p.lastSource] ?? 0) + 1;
        return acc;
      }, {} as Record<string, number>),
      maxAgeMs: snapshot.length
        ? Math.max(...snapshot.map(p => now - p.lastUpdatedAt))
        : 0,
    };

    res.json({
      ok: true,
      phase: '8.8.4-IA-PRICE-CACHE',
      summary,
      prices: snapshot.map(p => ({
        symbol: p.symbol,
        price: p.price,
        lastSource: p.lastSource,
        ageMs: now - p.lastUpdatedAt,
      })),
    });
  } catch (err) {
    res.status(500).json({
      ok: false,
      phase: '8.8.4-IA-PRICE-CACHE',
      error: (err as Error).message,
    });
  }
});
