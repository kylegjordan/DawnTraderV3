-- Drop old index if it exists
DROP INDEX IF EXISTS rtb_signals_symbol_strategy_idx;

-- Remove duplicates that block index creation
DELETE FROM rtb_signals a
USING rtb_signals b
WHERE a.ctid < b.ctid
  AND a.mode = b.mode
  AND a.symbol = b.symbol
  AND a.currency = b.currency
  AND a.strategy = b.strategy;

-- Recreate proper unique index at pair level
CREATE UNIQUE INDEX rtb_signals_symbol_currency_strategy_idx
ON public.rtb_signals (mode, symbol, currency, strategy);


✅ What this does

Enforces one unique signal per (mode, symbol, currency, strategy).

Prevents duplication across different statuses.

Fixes the refresh duplication errors.
