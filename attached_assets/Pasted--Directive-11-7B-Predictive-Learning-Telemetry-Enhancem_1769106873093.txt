🧭 Directive 11.7B — Predictive Learning Telemetry Enhancement

Phase: 11.7
Series: Predictive Learning Bridge (9 → 12)
Author: DawnTrader Quantitative Systems Architecture Team
Last Updated: 2026-01-24
Status: ✅ Approved for Immediate Replit Implementation

1. Objective

Bridge the deterministic Predictive Learning System (Phases 9–10) with real VTS outcome and skip data, enabling Adaptive Expectancy and Predictive Confidence to self-evolve continuously — even during passive simulation mode — while preserving strict live-data isolation.

This directive closes the Phase 9 → 12 learning loop by feeding regime- and strategy-level results from simulated trades into the telemetry system used by the predictive scoring engine.

2. Purpose & Scope

Enhancements introduced:

Extend telemetry ingestion to include:

/logs/virtual_trades/ → executed simulated trades

/logs/vts_skipped_signals/ → filtered/skipped signals

Compute per-regime × strategy metrics:

winRate, avgPnL %, skipRatio, and illiquidRatio

Persist hourly snapshots and manifest entries for traceability.

Increment telemetry version each cycle for downstream detection.

Tag all telemetry from this process with source: "VTS" to prevent cross-contamination.

Guarantee full read/write isolation from live or paper telemetry.

Forward compatibility:
Phase 12’s ML ingestion pipeline will use this same "VTS" source tag to differentiate simulated predictive telemetry from execution telemetry.

🧩 Task 1 — Telemetry Aggregator Extension

File: /server/core/logging/telemetry.ts

export async function updateRegimePerformanceFromVTS() {
  const executedDir = "/logs/virtual_trades";
  const skippedDir  = "/logs/vts_skipped_signals";
  const metrics = {};
  const metricsSkipped = {};
  let totalProcessed = 0;

  if (!fs.existsSync(executedDir) || !fs.existsSync(skippedDir)) return;

  // --- Executed Trades ---
  for (const file of fs.readdirSync(executedDir)) {
    const path = `${executedDir}/${file}`;
    const stat = fs.statSync(path);
    if (Date.now() - stat.mtimeMs > 7 * 24 * 60 * 60 * 1000) continue; // ignore >7 days old
    try {
      const data = JSON.parse(fs.readFileSync(path, "utf8"));
      if (!Array.isArray(data)) continue;
      for (const t of data) {
        const { regime, strategy } = t;
        const netProfitPercent = Number(t.netProfitPercent ?? 0);
        if (!regime || !strategy) continue;
        metrics[regime] ??= {};
        metrics[regime][strategy] ??= { wins:0, total:0, pnl:0 };
        metrics[regime][strategy].total++;
        if (netProfitPercent > 0) metrics[regime][strategy].wins++;
        metrics[regime][strategy].pnl += netProfitPercent * 100; // normalize %
        totalProcessed++;
      }
    } catch {
      logger.warn(`[11.7B][Telemetry] Skipping invalid JSON: ${file}`);
      continue;
    }
  }

  // --- Skipped Signals ---
  for (const file of fs.readdirSync(skippedDir)) {
    const path = `${skippedDir}/${file}`;
    try {
      const data = JSON.parse(fs.readFileSync(path, "utf8"));
      if (!Array.isArray(data)) continue;
      for (const s of data) {
        const { regime, reason } = s;
        if (!regime) continue;
        metricsSkipped[regime] ??= { lowROI:0, illiquid:0, total:0 };
        metricsSkipped[regime].total++;
        if (reason === "Low_ROI") metricsSkipped[regime].lowROI++;
        if (reason === "Illiquid_USD") metricsSkipped[regime].illiquid++;
      }
    } catch {
      logger.warn(`[11.7B][Telemetry] Skipping invalid JSON: ${file}`);
      continue;
    }
  }

  if (totalProcessed === 0) {
    logger.info("[11.7B][Telemetry] No new entries found, skipping snapshot.");
    return;
  }

  if (telemetry.source && telemetry.source !== "VTS") {
    logger.warn("[11.7B][Telemetry] Cross-source write attempt blocked.");
    return;
  }

  await telemetryLock.runExclusive(() => {
    for (const regime in metrics) {
      telemetry.regimePerformance[regime] ??= {};
      for (const strategy in metrics[regime]) {
        const m = metrics[regime][strategy];
        const s = metricsSkipped[regime] || {};
        telemetry.regimePerformance[regime][strategy] = {
          source: "VTS",
          winRate: m.wins / (m.total || 1),
          avgPnL:  m.pnl / (m.total || 1),
          skipRatio: (s.lowROI || 0) / (s.total || 1),
          illiquidRatio: (s.illiquid || 0) / (s.total || 1),
          updatedAt: new Date().toISOString()
        };
      }
    }

    telemetry.version = (telemetry.version ?? 0) + 1; // version increment
  });

  saveTelemetrySnapshot(totalProcessed, metrics);
}

🧩 Task 2 — Snapshot Persistence & Manifest
export function saveTelemetrySnapshot(totalProcessed, metrics) {
  fs.mkdirSync("/logs/telemetry", { recursive: true });
  const date = new Date().toISOString().slice(0,10);
  const regimeCount = Object.keys(metrics).length;
  const filename = `/logs/telemetry/regime_performance_${date}_VTS_${totalProcessed}.json`;

  fs.writeFileSync(filename, JSON.stringify(telemetry.regimePerformance, null, 2));

  const manifestPath = "/logs/telemetry/regime_performance_manifest.json";
  const manifest = fs.existsSync(manifestPath)
    ? JSON.parse(fs.readFileSync(manifestPath, "utf8")) : [];
  manifest.push({
    filename,
    totalProcessed,
    regimes: regimeCount,
    version: telemetry.version ?? 1,
    timestamp: new Date().toISOString()
  });
  fs.writeFileSync(manifestPath, JSON.stringify(manifest, null, 2));

  logger.info(`[11.7B][Telemetry] Snapshot saved (${totalProcessed} entries, ${regimeCount} regimes) → ${filename}`);
}
