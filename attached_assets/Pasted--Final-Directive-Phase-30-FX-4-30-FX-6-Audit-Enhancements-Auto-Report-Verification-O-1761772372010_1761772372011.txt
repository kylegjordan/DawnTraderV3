import { db } from "../db";
import axios from "axios";

export async function startLottieOversight() {
  console.log("[30.FX.4] Lottie oversight service initialized ✅");

  async function checkDHMA() {
    try {
      const username = "testuser123";
      const password = "SecurePass123!";
      const auth = await axios.post("http://localhost:5000/api/auth/login", { username, password });
      const token = auth.data.accessToken;

      const res = await axios.get("http://localhost:5000/api/strategy/dhma/telemetry?mode=live", {
        headers: { Authorization: `Bearer ${token}` },
      });

      const { hitRate, avgToxicity, avgSpreadTicks, entries } = res.data;
      const healthy = hitRate >= 0.45 && avgToxicity <= 0.75 && avgSpreadTicks <= 6 && entries >= 5;
      const status = healthy ? "active" : "suspended";
      const reason = healthy ? "ok" : "metrics_out_of_bounds";

      await db.query(
        `INSERT INTO trading_audit_log (event, strategy, status, reason)
         VALUES ('lottie_oversight', 'dhma', $1, $2)`,
        [status, reason]
      );

      console.log(`[Lottie Oversight] DHMA ${healthy ? "ACTIVE ✅" : "SUSPENDED ⚠️"}`);
    } catch (err) {
      console.error("[Lottie Oversight] Error:", err.message);
    }
  }

  // Every 5 minutes
  setInterval(checkDHMA, 5 * 60 * 1000);
  checkDHMA();
}

Phase 30.FX.6 – Cross-Strategy Optimization

File: server/services/latti-manager.ts

import { db } from "../db";
import axios from "axios";

export async function optimizeStrategyMix() {
  console.log("[30.FX.6] Running hourly optimization...");
  const strategies = ["dhma", "quantflow", "trendpulse", "volsurf", "momentumx"];
  const username = "testuser123";
  const password = "SecurePass123!";
  const auth = await axios.post("http://localhost:5000/api/auth/login", { username, password });
  const token = auth.data.accessToken;

  const telemetry = [];
  for (const s of strategies) {
    try {
      const { data } = await axios.get(`http://localhost:5000/api/strategy/${s}/telemetry?mode=live`, {
        headers: { Authorization: `Bearer ${token}` },
      });
      telemetry.push({ name: s, pnl: data.avgPLPerTrade || 0, winRate: data.hitRate || 0 });
    } catch {
      console.warn(`[30.FX.6] Skipped ${s}: telemetry unavailable`);
    }
  }

  telemetry.sort((a, b) => b.pnl - a.pnl);
  const top = telemetry.slice(0, 2);
  const bottom = telemetry.slice(-2);

  for (const s of top)
    await db.query(`INSERT INTO strategy_mix_log (strategy, old_weight, new_weight, reason)
                    VALUES ($1, NULL, 1.1, 'top performer')`, [s.name]);
  for (const s of bottom)
    await db.query(`INSERT INTO strategy_mix_log (strategy, old_weight, new_weight, reason)
                    VALUES ($1, NULL, 0.9, 'underperformer')`, [s.name]);

  console.log(`[30.FX.6] Mix updated: +10% for ${top.map(t => t.name).join(", ")}, -10% for ${bottom.map(b => b.name).join(", ")}`);
}

// Schedule hourly
setInterval(optimizeStrategyMix, 60 * 60 * 1000);

import express from "express";
import { db } from "../db";
const router = express.Router();

router.get("/api/system/health", async (_, res) => {
  const { rows: audits } = await db.query(
    "SELECT event, status, created_at FROM trading_audit_log ORDER BY created_at DESC LIMIT 3"
  );
  const { rows: mix } = await db.query(
    "SELECT strategy, new_weight, created_at FROM strategy_mix_log ORDER BY created_at DESC LIMIT 3"
  );
  res.json({
    uptime: process.uptime(),
    lastAuditEvents: audits,
    lastStrategyRebalances: mix,
    status: "healthy",
  });
});

export default router;

function ema(prev, next, alpha = 0.2) {
  return prev * (1 - alpha) + next * alpha;
}

import fs from "fs";
import { db } from "../db";

export async function generatePhase30Report() {
  const { rows: oversight } = await db.query(
    "SELECT * FROM trading_audit_log ORDER BY created_at DESC LIMIT 20"
  );
  const { rows: mix } = await db.query(
    "SELECT * FROM strategy_mix_log ORDER BY created_at DESC LIMIT 20"
  );

  const report = `
# Phase 30.FX.4–6 Report
Generated: ${new Date().toISOString()}

## Lottie Oversight Log
${oversight.map(o => `- ${o.created_at}: ${o.status} (${o.reason})`).join("\n")}

## Strategy Mix Changes
${mix.map(m => `- ${m.created_at}: ${m.strategy} → ${m.new_weight} (${m.reason})`).join("\n")}
`;
  fs.writeFileSync("./audit/phase30-fx4-6-report.md", report);
  console.log("✅ Report generated: audit/phase30-fx4-6-report.md");
}

// Auto-trigger on server start
generatePhase30Report();
