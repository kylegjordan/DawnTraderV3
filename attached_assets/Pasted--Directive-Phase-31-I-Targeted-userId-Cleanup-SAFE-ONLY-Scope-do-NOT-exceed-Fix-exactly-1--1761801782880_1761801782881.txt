4.2 CI grep check (fail build on new hardcoded UUIDs)

Add script scripts/check-no-hardcoded-uuids.sh:

#!/usr/bin/env bash
set -euo pipefail
if rg -n "\b[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b" \
  --type ts --type tsx --type js --type jsx --ignore-file .gitignore \
  | rg -v "fixtures|seeds|testdata|migrations" | tee /dev/stderr | grep .
then
  echo "❌ Hardcoded UUIDs detected. See matches above."
  exit 1
fi
echo "✅ No hardcoded UUIDs."
