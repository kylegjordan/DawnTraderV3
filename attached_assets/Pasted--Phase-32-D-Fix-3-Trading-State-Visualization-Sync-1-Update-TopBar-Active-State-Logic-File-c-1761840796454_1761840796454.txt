// Phase 32.D-Fix.3 — Unified Active State
const isActive = (() => {
  if (currentMode === 'paper') {
    return (
      tradingStatus?.isEngineActivePaper ||
      paperSimStatus?.isRunning ||
      false
    );
  }
  if (currentMode === 'live') {
    return tradingStatus?.isEngineActiveLive || false;
  }
  return false;
})();
