// New service that requests macro + sentiment analysis using GPT
import OpenAI from "openai";
import { MarketSnapshot } from "../types";

export async function getAIMarketAnalysis(snapshot: MarketSnapshot) {
  const client = new OpenAI();
  const prompt = `Analyze current market data...`;
  const response = await client.responses.create({
    model: "gpt-4o-mini",
    input: prompt
  });
  return response.output_text;
}
