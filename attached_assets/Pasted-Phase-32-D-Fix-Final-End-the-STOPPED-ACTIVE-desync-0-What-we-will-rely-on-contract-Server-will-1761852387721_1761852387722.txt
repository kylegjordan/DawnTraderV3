2b) use-trading.tsx — exported “active” (replace Lines 243–244)

Introduce a small helper and use it:

function deriveIsActive(state?: TradingState, paperSimStatus?: { isRunning?: boolean }) {
  // Authoritative boolean first
  if (state && typeof state.active === 'boolean') return state.active;
  if (!state) return !!paperSimStatus?.isRunning;                 // best-effort on first paint
  if (state.mode === 'paper') return !!(state.isEngineActivePaper || paperSimStatus?.isRunning);
  return !!state.isEngineActiveLive;
}

const isTradingActivePaper = deriveIsActive(tradingStatus, paperSimStatus);
const isTradingActiveLive  = deriveIsActive(tradingStatus, undefined); // same truth, per-mode UI can still read this


Also export a single isTradingActive to make Top Bar simpler:

return {

  isTradingActive,
