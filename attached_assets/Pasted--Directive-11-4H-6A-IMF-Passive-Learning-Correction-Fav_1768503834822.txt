export function getFavoredStrategiesForRegime(regime: string): string[] {
  switch (regime) {
    case "BULL_STABLE":
      return ["SMA Trend Ride", "VWAP Pullback", "Morning Star / Evening Star"];
    case "BEAR_VOLATILE":
      return ["Breakdown Pullback", "Counter-Reversal", "Fast Exit Short"];
    case "LOW_VOL_CHOP":
      return ["Range Trade", "Support Bounce", "Mean Reversion"];
    case "HIGH_VOL_IMPULSE":
      return ["Breakout", "Adaptive Flow", "VWAP Bounce"];
    case "TRANSITION":
      return ["Pivot Shift", "Morning Star / Evening Star"];
    default:
      return ["Range Trade"];
  }
}

export function getFavoredSignalTypesForRegime(regime: string): string[] {
  switch (regime) {

    case "BEAR_VOLATILE":

    case "LOW_VOL_CHOP":
    case "HIGH_VOL_IMPULSE":
      return ["Quantitative", "Hybrid"];
    case "TRANSITION":
      return ["Hybrid", "Pattern"];
    default:
      return ["Quantitative"];
  }
}

Step 3A — Create Historical Fallback for Passive Learning
// /server/core/metrics/imf-metrics.ts
export async function calculateIMFMetrics(symbol: string, context: any, isPassive: boolean) {
  let ohlcData = [];

  if (isPassive) {
    ohlcData = await getHistoricalCache(symbol, { limit: 100 }) || [];
  } else {
    ohlcData = await getLiveOHLC(symbol, { limit: 100 });
  }

  if (!ohlcData.length) {
    return { LQ: 0, VolNoise: 1.0, Correlation: 1.0, passesMetricFilter: false };
  }

  const LQ = calculateLogLiquidity(ohlcData);
  const VolNoise = calculateVolNoise(ohlcData);
  const Correlation = calculateCorrelation(ohlcData);
  const passesMetricFilter = LQ >= LQ_MIN && VolNoise <= VN_MAX && Correlation <= CORR_MAX;

  return { LQ, VolNoise, Correlation, passesMetricFilter };
}

Step 3B — Integrate Into FX5 Scanner

}

Step 4A — Replace Dual Scan Invocation

Locate around lines 240–250:

this.scanMode('live').catch(...)


Replace with:

const activeMode = context?.engine?.mode || 'paper';
const tradingActive = context?.engine?.tradingActive ?? false;

logger.info(`[11.4H.6A][ModeCheck] Mode=${activeMode} | TradingActive=${tradingActive}`);

if (!tradingActive) {
  logger.info(`[11.4H.6A][PassiveScan] Passive learning active — running single scan for mode=${activeMode}`);
  await this.scanMode(activeMode).catch((err) =>
    logger.error(`[11.4H.6A][PassiveScan][Error] ${err.message}`)
  );
} else {
  if (activeMode === 'live') {
    await this.scanMode('live').catch((err) =>
      logger.error(`[11.4H.6A][LiveScan][Error] ${err.message}`)
    );
  } else if (activeMode === 'paper') {
    await this.scanMode('paper').catch((err) =>
      logger.error(`[11.4H.6A][PaperScan][Error] ${err.message}`)
    );
  }
}
