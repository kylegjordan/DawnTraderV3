#!/bin/bash
TOKEN=$(curl -s -X POST http://localhost:5000/api/auth/login \
 -H "Content-Type: application/json" \
 -d '{"username":"testuser123","password":"SecurePass123!"}' | jq -r '.accessToken')

for ep in status goals portfolio telemetry; do
  t=$(curl -s -w "%{time_total}" -o /dev/null \
  -H "Authorization: Bearer $TOKEN" http://localhost:5000/api/$ep)
  echo "$ep,$t" >> logs/phase3c-api-latency.csv
done
