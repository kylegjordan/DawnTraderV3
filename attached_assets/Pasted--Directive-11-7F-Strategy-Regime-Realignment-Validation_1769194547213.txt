Weighted and smoothed DriftScore formula:

File: /server/core/analytics/mapping-drift-calculator.ts

import { DRIFT_CANONICAL } from "../../config/drift-definitions";
import { ema } from "../../utils/smoothing";

export function computeDriftScore(actualVolZ: number[], actualTrendZ: number[], regime: string): number {
  const ideal = DRIFT_CANONICAL[regime];
  if (!ideal) return 0;

  const smoothedVol = ema(actualVolZ, 0.4).at(-1) ?? actualVolZ.at(-1);
  const smoothedTrend = ema(actualTrendZ, 0.4).at(-1) ?? actualTrendZ.at(-1);

  const dVol = smoothedVol - ideal.idealVolZ;
  const dTrend = smoothedTrend - ideal.idealTrendZ;

  const score = Math.sqrt(
    ideal.weightVol * (dVol ** 2) + ideal.weightTrend * (dTrend ** 2)
  );

  return Math.min(Math.max(score, 0), 3);
}


✅ Uses weighted Euclidean distance between actual and ideal regime conditions
✅ Applies 3-window exponential smoothing to stabilize readings
✅ Interpretable range 0–3 (low = aligned, high = drifted)
