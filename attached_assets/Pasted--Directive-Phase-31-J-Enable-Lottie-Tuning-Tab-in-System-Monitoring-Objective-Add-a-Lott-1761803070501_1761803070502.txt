Create client/src/components/monitoring/lottie-tuning-tab.tsx

import React, { useEffect, useState } from "react";
import { Card, CardContent } from "@/components/ui/card";
import { LineChart, Line, XAxis, YAxis, Tooltip, ResponsiveContainer } from "recharts";

export default function LottieTuningTab() {
  const [data, setData] = useState<any>(null);

  async function fetchData() {
    const res = await fetch("/api/system/latti-tuning");
    const json = await res.json();
    setData(json);
  }

  useEffect(() => {
    fetchData();
    const interval = setInterval(fetchData, 30000);
    return () => clearInterval(interval);
  }, []);

  if (!data) return <div>Loading Lottie tuning metrics...</div>;

  return (
    <Card className="p-4">
      <CardContent>
        <h2 className="text-xl font-bold mb-2">Lottie Tuning Overview</h2>
        <div className="grid grid-cols-2 gap-4">
          <div>
            <p><strong>Last Run:</strong> {data.lastRun}</p>
            <p><strong>Confidence:</strong> {data.confidence}</p>
            <p><strong>Stability Score:</strong> {data.stabilityScore}</p>
            <p><strong>Tuning Cycle:</strong> {data.tuningCycle} min</p>
          </div>
          <div>
            <ResponsiveContainer width="100%" height={150}>
              <LineChart data={Object.entries(data.adjustments).map(([k, v]) => ({ name: k, value: v }))}>
                <XAxis dataKey="name" hide />
                <YAxis domain={[-0.05, 0.05]} />
                <Tooltip />
                <Line type="monotone" dataKey="value" stroke="#0ea5e9" />
              </LineChart>
            </ResponsiveContainer>
          </div>
        </div>
        <div className="mt-3 text-sm text-gray-400 italic">
          {data.passiveLearning ? "Observation Mode Active" : "Live Adaptive Mode"}
        </div>
      </CardContent>
    </Card>
  );
}
