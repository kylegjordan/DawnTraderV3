🧠 Phase 30.FX.3 — LATTI Adaptive Tuning for DHMA
A. Create server/services/dhma-tuning-service.ts
import { db } from "../db";
import { sql } from "drizzle-orm";
import { logger } from "../utils/logger";

export class DHMATuningService {
  static async processTelemetry(mode: "paper" | "live", telemetry: any) {
    try {
      const { hitRate, avgPLPerTrade, avgSpreadTicks, avgToxicity } = telemetry;

      const targetHit = 0.58;
      const minPL = 0.004;

      const rows = await db.execute(sql`
        SELECT key, value, min, max, step
        FROM strategy_param_schema
        WHERE strategy_type='dhma' AND trading_mode=${mode};
      `);

      const updates: { key: string; newValue: number }[] = [];

      const spread = rows.find((r: any) => r.key === "spreadMaxTicks");
      if (spread) {
        if (avgSpreadTicks > 5)
          updates.push({ key: "spreadMaxTicks", newValue: Math.max(spread.min, spread.value - spread.step) });
        else if (avgSpreadTicks < 3)
          updates.push({ key: "spreadMaxTicks", newValue: Math.min(spread.max, spread.value + spread.step) });
      }

      const tox = rows.find((r: any) => r.key === "toxicityLimit");
      if (tox) {
        if (avgToxicity > 0.75)
          updates.push({ key: "toxicityLimit", newValue: Math.max(tox.min, tox.value - tox.step) });
        else if (avgToxicity < 0.5)
          updates.push({ key: "toxicityLimit", newValue: Math.min(tox.max, tox.value + tox.step) });
      }

      const micro = rows.find((r: any) => r.key === "micropriceDeviation");
      if (micro) {
        if (hitRate < targetHit)
          updates.push({ key: "micropriceDeviation", newValue: Math.max(micro.min, micro.value - micro.step) });
        else if (hitRate > targetHit + 0.1)
          updates.push({ key: "micropriceDeviation", newValue: Math.min(micro.max, micro.value + micro.step) });
      }

      const flow = rows.find((r: any) => r.key === "signedFlowRatio");
      if (flow && avgPLPerTrade < minPL)
        updates.push({ key: "signedFlowRatio", newValue: Math.min(flow.max, flow.value + flow.step) });

      if (!updates.length) return;

      for (const u of updates) {
        await db.execute(sql`
          UPDATE strategy_param_schema
          SET value=${u.newValue}, updated_at=NOW()
          WHERE strategy_type='dhma' AND trading_mode=${mode} AND key=${u.key};
        `);

        await db.execute(sql`
          INSERT INTO trading_audit_log (strategy_type, trading_mode, event_type, details)
          VALUES ('dhma', ${mode}, 'parameter_update', ${JSON.stringify(u)});
        `);

        logger.info(`[DHMATuning] ${u.key} adjusted to ${u.newValue.toFixed(4)} (${mode})`);
      }
    } catch (err) {
      logger.error("DHMATuningService error", err);
    }
  }
}
