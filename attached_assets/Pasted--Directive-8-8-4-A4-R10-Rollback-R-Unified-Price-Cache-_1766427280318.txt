1️⃣ server/services/price-cache.ts — Replace Entire File
/**
 * Directive 8.8.4-A4.R10R — Unified, Rate-Governed Price Cache
 * 
 * This service consolidates all price retrieval logic for:
 * - Open trades
 * - Ready-to-Buy signals
 * - FX5 snapshots
 * 
 * Responsibilities:
 *  - Manage three priority buckets with distinct refresh intervals.
 *  - Maintain Kraken API load below 10 weighted requests/second.
 *  - Provide universal subscription/unsubscription API.
 *  - Deliver cached prices to all consuming modules.
 */

import { delay } from '../utils/delay';
import { logger } from '../utils/logger';
import { fetchKrakenTickers } from '../connectors/kraken-api';

interface CacheBucket {
  type: 'openTrade' | 'readyToBuy' | 'fx5Snapshot';
  symbols: Set<string>;
  refreshIntervalMs: number;
  lastRefresh: number;
}

const buckets: CacheBucket[] = [
  { type: 'openTrade', symbols: new Set(), refreshIntervalMs: 2000, lastRefresh: 0 },
  { type: 'readyToBuy', symbols: new Set(), refreshIntervalMs: 15000, lastRefresh: 0 },
  { type: 'fx5Snapshot', symbols: new Set(), refreshIntervalMs: 30000, lastRefresh: 0 }
];

const cache: Map<string, any> = new Map();
let currentWeight = 0;
let refreshing = false;

// Reset API weight counter every second
setInterval(() => (currentWeight = 0), 1000);

async function safeFetch(weight: number, fn: () => Promise<any>): Promise<any> {
  if (currentWeight + weight > 10) {
    await delay(250);
    return safeFetch(weight, fn);
  }
  currentWeight += weight;
  return fn();
}

async function refreshBuckets() {
  if (refreshing) return;
  refreshing = true;

  const now = Date.now();

  for (const bucket of buckets) {
    if (now - bucket.lastRefresh >= bucket.refreshIntervalMs && bucket.symbols.size > 0) {
      const symbols = Array.from(bucket.symbols);
      const batchSize = 100;
      for (let i = 0; i < symbols.length; i += batchSize) {
        const batch = symbols.slice(i, i + batchSize);
        await safeFetch(1, async () => {
          const data = await fetchKrakenTickers(batch);
          for (const [pair, ticker] of Object.entries(data)) {
            cache.set(pair, { ...ticker, updated: now });
          }
          logger.info(`[A4.R10R][PriceCache][${bucket.type}] refreshed ${batch.length} symbols`);
        });
      }
      bucket.lastRefresh = now;
    }
  }

  refreshing = false;
}

setInterval(refreshBuckets, 1000);

export function subscribe(symbol: string, bucketType: CacheBucket['type']) {
  const bucket = buckets.find(b => b.type === bucketType);
  if (!bucket) throw new Error(`Invalid bucket type: ${bucketType}`);
  bucket.symbols.add(symbol);
}

export function unsubscribe(symbol: string) {
  for (const b of buckets) b.symbols.delete(symbol);
}

export function getCachedPrice(symbol: string) {
  return cache.get(symbol);
}

// Log health snapshot once per minute
setInterval(() => {
  const open = buckets[0].symbols.size;
  const rtb = buckets[1].symbols.size;
  const fx5 = buckets[2].symbols.size;
  logger.info(`[A4.R10R][PriceCache][HEALTH] open=${open} rtb=${rtb} fx5=${fx5} weight=${currentWeight}/10`);
}, 60000);
