export function calculateEfficiencyRatio(prices: number[], window = 20): number {
  if (prices.length < window + 1) return 0;
  const change = Math.abs(prices[prices.length - 1] - prices[prices.length - window - 1]);
  const volatility = prices.slice(-window).reduce((sum, p, i, arr) =>
    i > 0 ? sum + Math.abs(p - arr[i - 1]) : sum, 0);
  return volatility === 0 ? 0 : +(change / volatility).toFixed(4);
}

C. Example Implementation (Final Version)
// server/utils/adaptive-kalman.ts
export class AdaptiveKalmanFilter {
  private x: number | null = null; // Detect first run
  private P = 1;

  update(price: number, ER: number, VolNoise: number): number {
    // 1. Cold Start: Seed with first price
    if (this.x === null) {
      this.x = price;
      console.log(`[9.3][INIT] Seeding Kalman with first price ${price}`);
      return price;
    }

    // 2. Adaptive R (1–50 range)
    const R = Math.max(1, Math.min(50, 1 + (1 - ER) * 50));

    // 3. Adaptive Q (scaled by VolNoise)
    const Q = Math.max(0.1, VolNoise * 0.5);

    // 4. Kalman Update
    const K = this.P / (this.P + R);
    this.x = this.x + K * (price - this.x);
    this.P = Math.max(1e-8, (1 - K) * this.P + Q);

    console.log(`[9.3][KALMAN] ER=${ER.toFixed(2)} R=${R.toFixed(2)} Q=${Q.toFixed(3)} x=${this.x.toFixed(4)}`);
    return this.x;
  }

  reset() {
    this.x = null;
    this.P = 1;
  }

  getState() {
    return this.x;
  }
}
