// Define benchmark symbols
const BENCHMARK_SYMBOLS = [
  'BTC/USD', 'BTC/USDT',
  'ETH/USD', 'ETH/USDT',
  'SOL/USD', 'SOL/USDT',
  'USDT/USD', 'USDC/USD', 'DAI/USD'
];

// --- Force-Include Logic ---
if (BENCHMARK_SYMBOLS.includes(symbol)) {
  pair.forceInclude = true;
  pair.benchmarkForceInclude = true;
  pair.poolType = 'BENCHMARK'; // Tag for UI
}

// --- Pool Assignment (Additive, Not Exclusive) ---
if (pair.benchmarkForceInclude) {
  benchmarkPool.push(pair);  // For Benchmark UI visibility
  idealPool.push(pair);      // Ensure VTS still scans & trades it
} else {
  idealPool.push(pair);
}


💡 Stablecoin nuance:
Even though stablecoins enter the pool, trade logic (e.g., range / volatility filters) ensures they won’t trigger signals unless they deviate significantly (> 0.05 %).
