✅ 2. Drift Computation Loop
// server/services/telemetry-aggregator.ts
import { computeDriftScore } from "../core/analytics/mapping-drift-calculator";

for (const [regime, strategies] of Object.entries(canonicalMap)) {
  driftScores[regime] = {};
  for (const strategy of strategies) {
    const z = aggregateZScores(pairTelemetry, regime, strategy);
    driftScores[regime][strategy] = computeDriftScore(z.volZHistory, z.trendZHistory, regime);
  }
}
