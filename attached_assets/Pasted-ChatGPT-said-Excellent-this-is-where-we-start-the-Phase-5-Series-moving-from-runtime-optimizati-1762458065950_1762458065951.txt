import zlib from "zlib";
const SAMPLE_RATE = 0.05; // 5 %

function shouldSample(){return Math.random() < SAMPLE_RATE;}

export function recordTelemetry(event){
  if(!shouldSample()) return;
  queue.push(event);
}

function flush(){
  if(queue.length===0)return;
  const json = JSON.stringify(queue.splice(0,50));
  const compressed = zlib.gzipSync(json);
  ws.broadcast("telemetry_batch",compressed);
  console.log(`[Gemini-5A] Flushed ${json.length}b → ${compressed.length}b`);
}
setInterval(flush,Number(process.env.TELEMETRY_BATCH_MS)||2000);
