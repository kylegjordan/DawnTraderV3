3️⃣ Implement Cascading Logic – scanner-service.ts

Create a token-bucket–guarded cascading scan system that builds on the existing getOHLC fetcher.

🔧 Optimization Applied:

✅ The scanTimeframe() function must be a wrapper/decorator around the existing getOHLC() fetch logic — not a rewrite.
It should perform a token-bucket check before calling getOHLC() and throttle if the request velocity exceeds the safe rate.

import { getOHLC } from './kraken-client'; // existing function

const tokenBucket = {
  tokens: TIMEFRAME_CONFIG.RATE_LIMITS.MAX_REQ_PER_SEC * TIMEFRAME_CONFIG.RATE_LIMITS.SAFETY_MARGIN,
  refillInterval: 1000, // per second
  lastRefill: Date.now(),
};

function consumeToken() {
  const now = Date.now();
  if (now - tokenBucket.lastRefill > tokenBucket.refillInterval) {
    tokenBucket.tokens = TIMEFRAME_CONFIG.RATE_LIMITS.MAX_REQ_PER_SEC * TIMEFRAME_CONFIG.RATE_LIMITS.SAFETY_MARGIN;
    tokenBucket.lastRefill = now;
  }
  if (tokenBucket.tokens <= 0) return false;
  tokenBucket.tokens--;
  return true;
}

export async function scanTimeframe(interval: Timeframe, pairs: string[] = ALL_PAIRS) {
  const results = [];
  for (const symbol of pairs) {
    if (!consumeToken()) {
      await new Promise(res => setTimeout(res, 100)); // throttle
    }
    const candles = await getOHLC(symbol, interval);
    results.push({ symbol, candles, timeframe: interval });
  }
  return results;
}

export async function cascadingScan() {
  const globalPairs = await scanTimeframe('1h');
  const activePairs = globalPairs.filter(p => p.regimeWeight > 0.5);

  const tacticalPairs = await scanTimeframe('15m', activePairs.map(p => p.symbol));
  const precisionPairs = await scanTimeframe('5m', tacticalPairs.map(p => p.symbol));

  return { globalPairs, tacticalPairs, precisionPairs };
}
