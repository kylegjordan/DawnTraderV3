Add helper in /server/config/regime-strategy-map.ts:

export function getTypeForStrategy(strategy: string): SignalType {
  const quant = ['vwap_pullback','sma_trend_ride','liquidity_trap','pivot_shift'];
  const pattern = ['abcd_long','support_bounce','triangle_breakout','double_bottom'];
  if (quant.includes(strategy)) return 'QUANT';
  if (pattern.includes(strategy)) return 'PATTERN';
  return 'HYBRID';
}

import { getTypeForStrategy } from "@/server/config/regime-strategy-map";

describe("Signal mapping verification", () => {
  const map = {
    breakout: "QUANT",
    abcd_long: "PATTERN",
    trend_flow: "HYBRID"
  };
  for (const [strategy, expected] of Object.entries(map)) {
    test(`${strategy} → ${expected}`, () => {
      expect(getTypeForStrategy(strategy)).toBe(expected);
    });
  }
