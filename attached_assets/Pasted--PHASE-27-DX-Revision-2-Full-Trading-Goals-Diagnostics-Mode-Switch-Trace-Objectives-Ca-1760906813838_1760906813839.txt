E — Ghost and Broken Reference Check
-- Sessions without context link
SELECT id, user_id, active, started_at
FROM paper_sim_sessions s
WHERE active=true
 AND NOT EXISTS (SELECT 1 FROM system_context c WHERE c.session_id=s.id);

-- Context rows pointing to missing session
SELECT user_id, session_id, trading_mode, is_engine_active
FROM system_context c
WHERE session_id IS NOT NULL
 AND NOT EXISTS (SELECT 1 FROM paper_sim_sessions s WHERE s.id=c.session_id);
