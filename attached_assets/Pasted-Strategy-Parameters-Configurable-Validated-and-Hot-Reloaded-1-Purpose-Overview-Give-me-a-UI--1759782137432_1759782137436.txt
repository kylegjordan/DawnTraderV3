Strategy Parameters — Configurable, Validated, and Hot-Reloaded
1) Purpose & Overview

Give me a UI + API to configure each strategy’s parameters (e.g., VWAP Pullback, ABCD Long, SMA Trend Ride) per user and per mode (live/paper). Persist changes, validate them, preview the impact, and have the trading engine consume the latest settings without a server restart. Log all changes (who/when/old→new) and allow safe rollback.

Key goals:

Mode isolation (independent settings for live vs paper).

Guardrails (server-side validation & sane defaults).

Immediate effect (engine reloads settings in memory).

Full audit (who changed what, when; AI vs user).

Ready for Goals Engine: settings can be adjusted by AI, but only after user approval.

Intent: Strong runtime validation of parameters per strategy.

import { z } from "zod";

export const BaseCommon = z.object({
  maxConcurrentPositions: z.number().int().min(0).max(20).default(5),
  riskPerTrade: z.number().min(0.0005).max(0.05).default(0.01),        // 0.05 = 5%
  takeProfitR: z.number().min(0.2).max(10).default(2),
  stopLossR: z.number().min(0.1).max(10).default(1),
  cooldownMinutes: z.number().int().min(0).max(240).default(5),
});

export const VWAP_PULLBACK_Schema = BaseCommon.extend({
  vwapLookbackMin: z.number().int().min(1).max(120).default(20),
  pullbackPct: z.number().min(0.001).max(0.05).default(0.01),
  minVolumeUsd: z.number().min(0).default(10000),
});

export const ABCD_LONG_Schema = BaseCommon.extend({
  minAtoBStrength: z.number().min(0.1).max(5).default(1.2),
  cPullbackPctMax: z.number().min(0.01).max(0.3).default(0.15),
  dBreakoutBufferPct: z.number().min(0).max(0.05).default(0.005),
});

export const SMA_TREND_Schema = BaseCommon.extend({
  fastSma: z.number().int().min(3).max(50).default(10),
  slowSma: z.number().int().min(10).max(200).default(50),
  trendStrengthMin: z.number().min(0).max(1).default(0.4),
});

export function getValidator(strategy: string) {
  switch (strategy) {
    case "VWAP_PULLBACK": return VWAP_PULLBACK_Schema;
    case "ABCD_LONG": return ABCD_LONG_Schema;
    case "SMA_TREND": return SMA_TREND_Schema;
    default: return BaseCommon;
  }
}

2.4 API endpoints

Intent: Mode-aware CRUD + dry-run validate + engine hot-reload trigger.

File: server/routes.ts

// GET current settings per strategy
app.get("/api/strategies/settings", authRequired, async (req, res) => {
  const userId = req.user.id;
  const mode = (String(req.query.mode) === "paper" ? "paper" : "live") as 'live'|'paper';
  const strategy = String(req.query.strategy);
  const row = await storage.getStrategySettings({ userId, mode, strategy });
  return res.json(row ?? {});
});

// GET all settings for a mode
app.get("/api/strategies/settings/all", authRequired, async (req, res) => {
  const userId = req.user.id;
  const mode = (String(req.query.mode) === "paper" ? "paper" : "live") as 'live'|'paper';
  const rows = await storage.listStrategySettings({ userId, mode });
  return res.json(rows);
});

// POST validate only (no save)
app.post("/api/strategies/settings/validate", authRequired, async (req, res) => {
  const { strategy, params } = req.body || {};
  const schema = getValidator(strategy);
  const parse = schema.safeParse(params);
  if (!parse.success) return res.status(400).json({ ok: false, errors: parse.error.flatten() });
  return res.json({ ok: true, normalized: parse.data });
});

// PUT save (validated) + audit + hot-reload
app.put("/api/strategies/settings", authRequired, async (req, res) => {
  const userId = req.user.id;
  const mode = (req.body?.mode === "paper" ? "paper" : "live") as 'live'|'paper';
  const strategy = String(req.body?.strategy);
  const schema = getValidator(strategy);
  const parse = schema.safeParse(req.body?.params);
  if (!parse.success) return res.status(400).json({ ok: false, errors: parse.error.flatten() });

  const prev = await storage.getStrategySettings({ userId, mode, strategy });
  const saved = await storage.upsertStrategySettings({
    userId, mode, strategy, params: parse.data
  });

  await storage.insertStrategySettingsAudit({
    userId, mode, strategy,
    prevParams: prev?.params ?? null,
    nextParams: saved.params,
    actorType: "user", actorId: req.user.username,
    reason: req.body?.reason || "manual update"
  });

  // Hot-reload into engine
  await EngineSettingsBus.publish({ userId, mode }); // see 2.5

  return res.json({ ok: true, saved });
});

2.5 Engine hot-reload (no restart)

Intent: Engine keeps a per-user/per-mode in-memory cache and refreshes on publish.

File: server/services/trading-engine.ts (or a dedicated bus)

// Simple in-process pub/sub
export const EngineSettingsBus = {
  _subs: new Set<(msg:{userId:string;mode:'live'|'paper'})=>void>(),
  subscribe(fn) { this._subs.add(fn); return () => this._subs.delete(fn); },
  publish(msg) { this._subs.forEach(fn => fn(msg)); },
};

// In engine bootstrap:
EngineSettingsBus.subscribe(async ({ userId, mode }) => {
  // Pull latest settings from storage and refresh engine memory
  const rows = await storage.listStrategySettings({ userId, mode });
  EngineState.setSettings(userId, mode, rows);   // your existing mem cache pattern
});

// Also: on engine init, load settings once:
const initial = await storage.listStrategySettings({ userId, mode });
EngineState.setSettings(userId, mode, initial);
