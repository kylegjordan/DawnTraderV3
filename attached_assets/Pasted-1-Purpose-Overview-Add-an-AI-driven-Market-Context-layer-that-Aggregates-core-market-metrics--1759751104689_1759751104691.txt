1) Purpose & Overview

Add an AI-driven Market Context layer that:

Aggregates core market metrics (trend, volatility, dominance, volume).

Calls GPT to classify the current regime (e.g., bullish / bearish / neutral / high-vol / low-vol / accumulation / distribution) with confidence and tactical recommendations.

Stores analyses per mode (live/paper) and exposes them via API.

Feeds strategy weighting (signal_weights) and appears on the Dashboard as a Market Insights card.

Is included in the Daily Brief.

Key requirements: mode isolation, low cost, safe fallbacks.

2.3 AI Market Analyzer service

Intent: Call OpenAI with a strict JSON-only prompt; parse and persist analysis.

Replit: Create server/services/ai-market-analyzer.ts.

// server/services/ai-market-analyzer.ts
import OpenAI from "openai";
import { getMarketSnapshot } from "./market-snapshot";
import { type InsertAiMarketAnalysis } from "../../shared/schema";
import storage from "../storage";

const MODEL = process.env.OPENAI_MODEL_MARKET || "gpt-4o-mini"; // cost-aware default

export async function runAiMarketAnalysis(mode: 'live'|'paper'): Promise<InsertAiMarketAnalysis> {
  const snapshot = await getMarketSnapshot();
  const prompt = `
You are a trading market analyst. Given the JSON snapshot of crypto market conditions, classify:
- regime: one of ["bullish","bearish","neutral","accumulation","distribution","high_volatility","low_volatility"]
- confidence: 0..100
- summary: 2-3 sentences in plain English
- recommendations: 3-5 concise bullet actions for strategy posture (e.g., risk per trade, number of concurrent positions, long/short bias)

Return ONLY valid JSON with keys: regime, confidence, summary, recommendations.

SNAPSHOT:
${JSON.stringify(snapshot)}
`;

  const client = new OpenAI();
  const res = await client.responses.create({
    model: MODEL,
    input: prompt,
    response_format: { type: "json_object" },
    max_output_tokens: 600,
  });

  // Defensive parse
  const text = (res as any)?.output_text || (res as any)?.output?.[0]?.content?.[0]?.text || "{}";
  let parsed: any = {};
  try { parsed = JSON.parse(text); } catch { parsed = {}; }

  const row: InsertAiMarketAnalysis = {
    date: new Date().toISOString().slice(0,10), // yyyy-mm-dd
    mode,
    regime: parsed.regime || "neutral",
    confidence: Number(parsed.confidence ?? 50),
    summary: parsed.summary || "Market context unavailable.",
    recommendations: Array.isArray(parsed.recommendations) ? parsed.recommendations : [],
    snapshot,
  };

  await storage.upsertAiMarketAnalysisByDateMode(row);
  return row;
}


Cost control: Uses gpt-4o-mini by default; short prompt; JSON output; capped tokens.

app.get("/api/market/ai-analysis", async (req, res) => {
  const mode = (req.query.mode as string) === 'paper' ? 'paper' : 'live';
  const row = await storage.getLatestAiMarketAnalysis(mode);
  if (!row) return res.status(404).json({ error: "No analysis yet" });

// GET history by date range
app.get("/api/market/ai-analysis/history", async (req, res) => {
  const mode = (req.query.mode as string) === 'paper' ? 'paper' : 'live';
  const { from, to } = req.query as { from?: string; to?: string };
  if (!from || !to) return res.status(400).json({ error: "from & to required (YYYY-MM-DD)" });

  res.json(rows);
});

2.6 Frontend – Market Insights Widget

Intent: Display regime, confidence, summary, and top recommendations; mode-aware.

Replit: Create client/src/components/goals/market-insights-widget.tsx.

// client/src/components/goals/market-insights-widget.tsx
import { Card } from "@/components/ui/card";
import { useQuery } from "@tanstack/react-query";
import { apiRequest } from "@/lib/query";
import { useTradingMode } from "@/contexts/trading-mode-context";

export default function MarketInsightsWidget() {
  const { mode } = useTradingMode();
  const { data, isLoading, error } = useQuery({
    queryKey: ["aiMarketAnalysis", mode],
    queryFn: () => apiRequest(`/api/market/ai-analysis?mode=${mode}`),
    staleTime: 10 * 60 * 1000,
  });

  if (isLoading) return <Card className="p-4">Loading market insights…</Card>;
  if (error || !data) return <Card className="p-4">No market analysis available.</Card>;

  const badgeColor =
    data.regime.includes("bull") ? "bg-green-100 text-green-800" :
    data.regime.includes("bear") ? "bg-red-100 text-red-800" :
    "bg-gray-100 text-gray-800";

  return (
    <Card className="p-4">
      <div className="flex justify-between items-start">
        <h3 className="text-sm font-semibold text-gray-500 uppercase tracking-wide">Market Insights</h3>
        <span className={`px-2 py-1 rounded ${badgeColor}`}>{data.regime} · {data.confidence}%</span>
      </div>
      <p className="mt-2 text-sm text-gray-700">{data.summary}</p>
      {Array.isArray(data.recommendations) && data.recommendations.length > 0 && (
        <ul className="mt-3 list-disc list-inside text-sm text-gray-700">
          {data.recommendations.slice(0,3).map((r: string, i: number) => <li key={i}>{r}</li>)}
        </ul>
      )}
    </Card>
  );
}
