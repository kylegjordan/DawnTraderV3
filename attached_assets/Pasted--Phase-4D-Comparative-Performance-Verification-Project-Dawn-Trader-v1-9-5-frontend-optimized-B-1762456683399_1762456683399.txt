4️⃣ Benchmark API Latency & Cache Efficiency
TOKEN=$(curl -s -X POST http://localhost:5000/api/auth/login \
 -H "Content-Type: application/json" \
 -d '{"username":"testuser123","password":"SecurePass123!"}' | jq -r '.accessToken')

for ep in status portfolio system/config diagnostics/scan telemetry/summary; do
  t=$(curl -s -w "%{time_total}" -o /dev/null \
  -H "Authorization: Bearer $TOKEN" http://localhost:5000/api/$ep)
  printf "%-22s %s sec\n" "$ep" "$t"
done | tee logs/phase4d-api-latency.csv

grep "\[Gemini-Profile\]" /tmp/logs/Start_application_* | tail -5 \
| tee logs/phase4d-cachehit.txt


✅ Targets:

Avg API ≤ 110 ms

Cache Hit ≥ 0.85
