🧩 Task 2 — Dynamic ROI Helper

File: /server/core/calculations/expectancy.ts

import { ROI_FLEX_MULTIPLIER, ROI_MIN, ROI_MAX } from "../../config/adaptive-thresholds";

export function getDynamicROIThreshold(regime: string, predictiveConfidence: number): number {
  const base = getMinROIForRegime(regime);
  const boundedConfidence = Math.min(Math.max(predictiveConfidence, 0.0), 1.0);
  const dynamicROI = base * (1 - (boundedConfidence - 0.5) * ROI_FLEX_MULTIPLIER);
  return Math.min(Math.max(dynamicROI, ROI_MIN), ROI_MAX);
}


✅ Confidence-driven thresholding.
✅ Hard ROI bounds (1–4%) for deterministic safety.

🧩 Task 3 — Profitability Gate (with Friction Awareness)

File: /server/core/calculations/expectancy.ts

import { DEFAULT_FEE, DEFAULT_SLIPPAGE } from "../../config/adaptive-thresholds";

export function isSignalProfitable(
  entryPrice: number,
  targetPrice: number,
  regime: string,
  predictiveConfidence?: number,
  fee: number = DEFAULT_FEE,
  estimatedSlippage: number = DEFAULT_SLIPPAGE
): boolean {
  const roi = (targetPrice - entryPrice) / Math.max(entryPrice, 1e-8);

  const dynamicROI = getDynamicROIThreshold(regime, predictiveConfidence ?? 0.5);

  // NEW: Friction-aware floor
  const estimatedFriction = (fee * 2) + estimatedSlippage;
  const requiredROI = Math.max(dynamicROI, estimatedFriction * 1.1); // adds 10% safety buffer

  return roi >= requiredROI;
}


✅ Prevents false positives in high-friction regimes.
✅ Ensures every accepted signal clears total transaction costs plus buffer.

🧩 Task 4 — PredictiveConfidence Source

File: /server/core/scoring/final-score.ts

export function getPredictiveConfidence(symbol: string, regime: string, strategy: string): number {
  const perf = telemetry.regimePerformance?.[regime]?.[strategy];
  if (!perf || perf.winRate == null) return 0.5; // neutral fallback

  const confidence = sigmoid((perf.winRate - 0.5) * 6);
  telemetry.regimePerformance[regime][strategy].predictiveConfidence = confidence;
  return Math.min(Math.max(confidence, 0.0), 1.0);
}


✅ Exports to telemetry snapshot for transparency.
✅ Confidence drives ROI scaling globally (VTS + SQE).
