Logging Function (with concurrency safety & schema)

import fs from "fs";
import path from "path";
import { safeAppendJSON } from "../utils/logging"; // ensures atomic writes

export function logPredictiveAdjustment(entry: {
  category: AdjustmentCategory;
  parameter: string;
  oldValue: number;
  newValue: number;
  regime?: string;
  strategy?: string;
  reason: string;
}) {
  const dir = "logs/predictive_adjustments";
  if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });
  const file = path.join(dir, `${new Date().toISOString().slice(0,10)}.json`);

  const impact =
    Math.abs(entry.newValue - entry.oldValue) /
    Math.max(Math.abs(entry.oldValue), 0.001);

  const record = {
    _schema: PREDICTIVE_ADJUSTMENTS_SCHEMA,
    timestamp: new Date().toISOString(),
    ...entry,
    delta: +(entry.newValue - entry.oldValue).toFixed(4),
    impact: +impact.toFixed(3),
  };

  safeAppendJSON(file, record); // atomic file-safe write

  fs.appendFileSync(
    "logs/system_events.log",
    `[11.7D.1][Adjustment] ${entry.category}:${entry.parameter} ${entry.oldValue}→${entry.newValue} (${entry.reason})\n`
  );
}

🧩 Task 5 — Concurrency Safety Utility

File: /server/utils/logging.ts

import fs from "fs";

export function safeAppendJSON(file, record) {
  const retries = 3;
  for (let i = 0; i < retries; i++) {
    try {
      const existing = fs.existsSync(file)
        ? JSON.parse(fs.readFileSync(file, "utf8"))
        : [];
      existing.push(record);
      fs.writeFileSync(file, JSON.stringify(existing, null, 2));
      return;
    } catch (err) {
      if (i === retries - 1) throw err;
      console.warn(`[safeAppendJSON] Retry ${i+1} for ${file}`);
    }
  }
}


✅ Prevents concurrent writes from corrupting JSON.
