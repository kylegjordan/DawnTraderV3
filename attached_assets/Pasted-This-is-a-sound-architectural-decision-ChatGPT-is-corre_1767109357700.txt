Target System: Dawn Trader v3.1 Infrastructure Objective: Upgrade BOTH WebSocket ingress points (Execution & Analytics) to Kraken v2 to ensure a unified, continuous data stream for all system components.

New (v2): wss://ws.kraken.com/v2 (Updates on BBO/Trades. Continuous).

2.1 Step 1: Create the Translation Utility
File: server/services/market-data/kraken-v2-translator.ts (New File) Action: Create this utility to ensure both adapters translate data identically.

// server/services/market-data/kraken-v2-translator.ts

export interface KrakenV2Update {
  symbol: string;
  ask?: number;
  bid?: number;
  last?: number;
  close?: number; // v2 sometimes uses 'close' or 'last'
  volume?: number;
  // Fallback for hybrid/legacy fields
  a?: string[]; 
  b?: string[]; 
  c?: string[];
}

/**
 * Translates Kraken v2 verbose objects into the compact v1 format
 * expected by Dawn Trader's internal pricing adapters.
 * Output: { a: [ask], b: [bid], c: [close] }
 */
export function translateV2ToV1(update: KrakenV2Update) {
  return {
    a: [String(update.ask || update.a?.[0] || 0)],
    b: [String(update.bid || update.b?.[0] || 0)],
    c: [String(update.last || update.close || update.c?.[0] || 0)]
  };
}
